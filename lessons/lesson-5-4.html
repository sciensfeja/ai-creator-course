Проблема галлюцинаций LLM

Почему ИИ выдумывает факты?
+100 XP

Большие языковые модели — это генераторы текста, а не базы знаний. Они предсказывают следующее слово на основе вероятностей. Когда модель не знает ответа, она всё равно генерирует текст — и часто выдумывает факты. Это называется галлюцинация.

🔍 Что такое RAG?

Retrieval-Augmented Generation
+200 XP

RAG (Retrieval-Augmented Generation) — это паттерн, где перед генерацией ответа модель сначала ищет релевантную информацию в базе знаний, а потом использует её как контекст для ответа. Это как поиск в Google + ChatGPT в одном флаконе!

📝 Как работает RAG: схема

# RAG Pipeline - Step by Step
# 
# 1. INDEXING PHASE (done once):
#    Documents → Split into chunks → Create embeddings → Store in vector DB
#
# 2. QUERY PHASE (done for each question):
#    User question → Create embedding → Search similar chunks → 
#    Combine question + chunks → Send to LLM → Get grounded answer

rag_pipeline = """
┌─────────────┐
│  Documents  │ (PDF, TXT, HTML)
└──────┬──────┘
       │
       ▼
┌─────────────┐
│   Chunking  │ (split into 500-token pieces)
└──────┬──────┘
       │
       ▼
┌─────────────┐
│  Embedding  │ (text → vector of 1536 numbers)
└──────┬──────┘
       │
       ▼
┌─────────────┐
│ Vector DB   │ (FAISS, ChromaDB, Pinecone)
└──────┬──────┘
       │
       ▼ (search)
┌─────────────┐
│   Top-K     │ (most similar chunks)
│   Chunks    │
└──────┬──────┘
       │
       ▼ (as context)
┌─────────────┐
│     LLM     │ (GPT-4, Claude, etc.)
└──────┬──────┘
       │
       ▼
┌─────────────┐
│   Answer    │ (grounded in facts)
└─────────────┘
"""
print(rag_pipeline)

Подготовка документов

Прежде чем искать по документам, их нужно разбить на чанки (кусочки) и превратить в эмбеддинги (векторы). Размер чанка критичен: слишком маленький — потеря контекста, слишком большой — шум в поиске.

📝 Чанкинг и создание эмбеддингов

import numpy as np
from openai import OpenAI
import os

client = OpenAI(api_key=os.getenv('OPENAI_API_KEY'))

# Sample document
document = """
Artificial intelligence (AI) is intelligence demonstrated by machines, 
as opposed to natural intelligence displayed by animals including humans. 
AI research has been defined as the field of study of intelligent agents, 
which refers to any system that perceives its environment and takes actions 
that maximize its chance of achieving its goals. The term "artificial 
intelligence" had previously been used to describe machines that mimic 
and display "human" cognitive skills that are associated with the human 
mind, such as "learning" and "problem-solving". This definition has since 
been rejected by major AI researchers who now describe AI in terms of 
rationality and acting rationally, which does not limit how intelligence 
can be articulated.
"""

# Step 1: Split document into chunks (simple character-based splitting)
def split_into_chunks(text, chunk_size=100, overlap=20):
    """Split text into overlapping chunks"""
    chunks = []
    start = 0
    while start < len(text):
        end = start + chunk_size
        chunk = text[start:end].strip()
        if chunk:
            chunks.append(chunk)
        start += chunk_size - overlap  # Move with overlap
    return chunks

chunks = split_into_chunks(document, chunk_size=150, overlap=30)
print(f"Created {len(chunks)} chunks:\n")
for i, chunk in enumerate(chunks):
    print(f"Chunk {i+1}: {chunk[:80]}...")
    print()

# Step 2: Create embeddings for each chunk
def create_embeddings(texts):
    """Create embeddings using OpenAI API"""
    response = client.embeddings.create(
        model="text-embedding-3-small",
        input=texts
    )
    return [item.embedding for item in response.data]

# For demo, let's create mock embeddings (in real app, use API above)
def mock_embeddings(texts):
    """Create random embeddings for demonstration"""
    return [np.random.randn(1536).tolist() for _ in texts]

chunk_embeddings = mock_embeddings(chunks)
print(f"Embedding dimension: {len(chunk_embeddings[0])}")
print(f"Total embeddings created: {len(chunk_embeddings)}")
⚠️ Правила чанкинга 1. Размер чанка: 200-500 токенов — оптимально для большинства задач
2. Перекрытие (overlap): 10-20% размера чанка, чтобы не терять контекст на границах
3. Границы: лучше резать по предложениям или абзацам, а не по символам
4. Метаданные: сохраняй источник каждого чанка (файл, страница) для цитирования

📝 Поиск с FAISS

import numpy as np

# FAISS - Facebook AI Similarity Search
# Install: pip install faiss-cpu
try:
    import faiss
    FAISS_AVAILABLE = True
except ImportError:
    FAISS_AVAILABLE = False
    print("FAISS not installed. Using numpy fallback for demo.")

class VectorStore:
    """Simple vector store with similarity search"""
    
    def __init__(self, dimension=1536):
        self.dimension = dimension
        self.chunks = []
        self.embeddings = []
        
        if FAISS_AVAILABLE:
            # Use FAISS for fast search
            self.index = faiss.IndexFlatIP(dimension)  # Inner product (cosine similarity)
        else:
            # Fallback to numpy
            self.index = None
    
    def add(self, chunks, embeddings):
        """Add chunks and their embeddings to the store"""
        self.chunks.extend(chunks)
        embeddings_array = np.array(embeddings, dtype='float32')
        
        # Normalize for cosine similarity
        norms = np.linalg.norm(embeddings_array, axis=1, keepdims=True)
        embeddings_array = embeddings_array / norms
        
        if self.index:
            self.index.add(embeddings_array)
        else:
            self.embeddings = embeddings_array.tolist()
        
        print(f"Added {len(chunks)} chunks to vector store")
    
    def search(self, query_embedding, top_k=3):
        """Find most similar chunks to query"""
        query_array = np.array([query_embedding], dtype='float32')
        norms = np.linalg.norm(query_array, axis=1, keepdims=True)
        query_array = query_array / norms
        
        if self.index:
            scores, indices = self.index.search(query_array, top_k)
            results = [(self.chunks[idx], scores[0][i]) 
                      for i, idx in enumerate(indices[0]) if idx != -1]
        else:
            # Numpy fallback: compute cosine similarity
            scores = []
            for i, emb in enumerate(self.embeddings):
                emb_array = np.array(emb, dtype='float32')
                similarity = np.dot(query_array[0], emb_array)
                scores.append((i, similarity))
            
            scores.sort(key=lambda x: x[1], reverse=True)
            results = [(self.chunks[idx], score) for idx, score in scores[:top_k]]
        
        return results

# Demo: Create vector store and search
store = VectorStore(dimension=1536)

# Add our chunks from previous level
store.add(chunks, chunk_embeddings)

# Search for relevant chunks
query = "What is artificial intelligence?"
query_embedding = np.random.randn(1536).tolist()  # Mock embedding

results = store.search(query_embedding, top_k=2)
print(f"\nQuery: '{query}'")
print(f"\nTop {len(results)} relevant chunks:")
for i, (chunk, score) in enumerate(results, 1):
    print(f"{i}. (Score: {score:.3f}) {chunk[:100]}...")

📝 Класс RAGPipeline

import os
from openai import OpenAI

class RAGPipeline:
    """Complete RAG pipeline implementation"""
    
    def __init__(self, openai_api_key=None):
        self.client = OpenAI(api_key=openai_api_key or os.getenv('OPENAI_API_KEY'))
        self.vector_store = VectorStore(dimension=1536)
        self.chunks = []
    
    def load_documents(self, texts):
        """Load and index documents"""
        print("📄 Loading documents...")
        all_chunks = []
        for text in texts:
            chunks = split_into_chunks(text, chunk_size=200, overlap=40)
            all_chunks.extend(chunks)
        
        print(f"✂️ Created {len(all_chunks)} chunks")
        
        # Create embeddings (use mock for demo)
        embeddings = mock_embeddings(all_chunks)
        
        # Add to vector store
        self.vector_store.add(all_chunks, embeddings)
        self.chunks = all_chunks
        
        print(f"✅ Indexed {len(all_chunks)} chunks")
    
    def retrieve(self, query, top_k=3):
        """Retrieve relevant chunks for query"""
        # Create query embedding
        query_embedding = mock_embeddings([query])[0]
        
        # Search vector store
        results = self.vector_store.search(query_embedding, top_k=top_k)
        
        # Extract chunks and scores
        retrieved_chunks = [chunk for chunk, score in results]
        return retrieved_chunks
    
    def generate_answer(self, query, retrieved_chunks):
        """Generate answer using LLM with retrieved context"""
        # Build context from retrieved chunks
        context = "\n\n".join([f"--- Chunk {i+1} ---\n{chunk}" 
                              for i, chunk in enumerate(retrieved_chunks)])
        
        # Create prompt with context
        prompt = f"""Answer the question based ONLY on the provided context. 
If the answer is not in the context, say "I don't know based on the provided information."

Context:
{context}

Question: {query}

Answer:"""
        
        # Call LLM (mock for demo)
        response = self.client.chat.completions.create(
            model="gpt-4o-mini",
            messages=[{"role": "user", "content": prompt}],
            temperature=0.3,  # Low temperature for factual answers
            max_tokens=300
        )
        
        return response.choices[0].message.content
    
    def ask(self, query):
        """Full RAG pipeline: retrieve + generate"""
        print(f"\n❓ Question: {query}")
        
        # Step 1: Retrieve relevant chunks
        print("🔍 Searching knowledge base...")
        retrieved_chunks = self.retrieve(query, top_k=3)
        
        print(f"📚 Found {len(retrieved_chunks)} relevant chunks")
        
        # Step 2: Generate answer
        print("🤖 Generating answer...")
        answer = self.generate_answer(query, retrieved_chunks)
        
        print(f"\n💡 Answer: {answer}")
        return answer

# Demo usage
rag = RAGPipeline()

# Load documents
documents = [
    "Python is a high-level programming language known for its simplicity and readability.",
    "JavaScript is primarily used for web development and runs in browsers.",
    "Machine learning is a subset of AI that learns from data without explicit programming."
]

rag.load_documents(documents)

# Ask questions
rag.ask("What is Python?")
rag.ask("Tell me about web development languages")

📝 RAG-приложение с загрузкой PDF

import os
import PyPDF2
from openai import OpenAI

class DocumentRAGApp:
    """Production-ready RAG application for PDF documents"""
    
    def __init__(self):
        self.client = OpenAI(api_key=os.getenv('OPENAI_API_KEY'))
        self.rag = RAGPipeline()
        self.documents = []
    
    def load_pdf(self, pdf_path):
        """Extract text from PDF file"""
        print(f"📄 Loading PDF: {pdf_path}")
        
        with open(pdf_path, 'rb') as file:
            pdf_reader = PyPDF2.PdfReader(file)
            text = ""
            for page in pdf_reader.pages:
                text += page.extract_text() + "\n"
        
        self.documents.append({
            'path': pdf_path,
            'text': text,
            'pages': len(pdf_reader.pages)
        })
        
        print(f"✅ Extracted {len(text)} characters from {len(pdf_reader.pages)} pages")
        return text
    
    def load_multiple_pdfs(self, pdf_paths):
        """Load multiple PDF documents"""
        all_texts = []
        for path in pdf_paths:
            text = self.load_pdf(path)
            all_texts.append(text)
        
        # Index all documents
        self.rag.load_documents(all_texts)
        print(f"\n📚 Indexed {len(pdf_paths)} documents")
    
    def ask_with_sources(self, query):
        """Ask question and get answer with source citations"""
        # Retrieve chunks
        retrieved_chunks = self.rag.retrieve(query, top_k=3)
        
        # Build context with sources
        context_parts = []
        for i, chunk in enumerate(retrieved_chunks):
            context_parts.append(f"[Source {i+1}]: {chunk}")
        
        context = "\n\n".join(context_parts)
        
        # Generate answer with citation requirement
        prompt = f"""You are a helpful assistant that answers questions based on provided sources.
Always cite which source you're using with [Source N] notation.
If the answer is not in the sources, say "I cannot find this information in the provided documents."

Sources:
{context}

Question: {query}

Answer with citations:"""
        
        response = self.client.chat.completions.create(
            model="gpt-4o-mini",
            messages=[{"role": "user", "content": prompt}],
            temperature=0.2
        )
        
        return response.choices[0].message.content
    
    def interactive_chat(self):
        """Interactive chat loop"""
        print("\n" + "="*60)
        print("🤖 RAG Document Assistant")
        print("="*60)
        print("Commands: 'quit' to exit, 'docs' to list documents")
        print("="*60 + "\n")
        
        while True:
            query = input("You: ").strip()
            
            if query.lower() == 'quit':
                print("Goodbye! 👋")
                break
            elif query.lower() == 'docs':
                print(f"\n📚 Loaded documents ({len(self.documents)}):")
                for i, doc in enumerate(self.documents, 1):
                    print(f"  {i}. {doc['path']} ({doc['pages']} pages)")
                continue
            elif not query:
                continue
            
            print("\n🤖 Thinking...")
            answer = self.ask_with_sources(query)
            print(f"\n{answer}\n")

# Run the application
if __name__ == "__main__":
    app = DocumentRAGApp()
    
    # Load sample PDFs (replace with your files)
    pdf_files = ["document1.pdf", "document2.pdf"]
    
    # Check if files exist
    existing_files = [f for f in pdf_files if os.path.exists(f)]
    
    if existing_files:
        app.load_multiple_pdfs(existing_files)
        app.interactive_chat()
    else:
        print("📝 No PDF files found. Create sample PDFs or update pdf_files list.")
        print("\nDemo mode: Using sample text documents")
        app.rag.load_documents([
            "Python is a versatile programming language used in AI, web development, and data science.",
            "Machine learning models require training data to learn patterns and make predictions.",
            "Neural networks are inspired by biological brains and consist of layers of neurons."
        ])
        app.interactive_chat()
