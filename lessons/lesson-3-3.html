🔗 Цепное правило (Chain Rule)

Математический фундамент backpropagation
+120 XP

Чтобы понять, как изменить каждый вес в сети, нам нужно знать, как ошибка зависит от этого веса. Но веса находятся глубоко в сети! Здесь на помощь приходит цепное правило из математического анализа: если y зависит от x через промежуточную переменную z, то dy/dx = (dy/dz) * (dz/dx).

Цепное правило в действии

import numpy as np

# Simple example: y = (2x + 3)^2
# We want to find dy/dx using chain rule

def forward(x):
    """Forward pass: compute y from x"""
    z = 2 * x + 3  # Intermediate variable
    y = z ** 2     # Output
    return z, y

def backward(x, z, y, dy=1.0):
    """Backward pass: compute gradients using chain rule"""
    # dy/dz = 2z
    dz = 2 * z * dy
    
    # dz/dx = 2
    dx = 2 * dz
    
    return dx, dz

# Test with x = 1
x = 1.0
z, y = forward(x)
dx, dz = backward(x, z, y)

print(f"x = {x}")
print(f"z = 2x + 3 = {z}")
print(f"y = z^2 = {y}")
print(f"\nGradients (chain rule):")
print(f"dy/dz = 2z = {dz}")
print(f"dz/dx = 2")
print(f"dy/dx = (dy/dz) * (dz/dx) = {dx}")

# Verify with numerical gradient
h = 0.0001
numerical_grad = ((2*(x+h)+3)**2 - (2*(x-h)+3)**2) / (2*h)
print(f"\nNumerical verification: dy/dx ≈ {numerical_grad:.4f}")
💡 Почему это важно? Цепное правило позволяет нам "протаскивать" градиент ошибки через все слои сети, от выхода ко входу. Это и есть обратное распространение!

️ Backpropagation вручную

Вычисляем градиенты для простой сети
+150 XP

Давай реализуем backpropagation для простой сети: вход → скрытый слой (2 нейрона) → выход. Мы вручную вычислим все градиенты и обновим веса.

Ручной backpropagation

import numpy as np

# Simple network: 1 input -> 2 hidden -> 1 output
# Using sigmoid activation

def sigmoid(z):
    return 1 / (1 + np.exp(-z))

def sigmoid_derivative(s):
    return s * (1 - s)

# Initialize weights (small random values)
np.random.seed(42)
w1 = np.random.randn(1, 2) * 0.5  # Input to hidden
b1 = np.zeros((1, 2))
w2 = np.random.randn(2, 1) * 0.5  # Hidden to output
b2 = np.zeros((1, 1))

# Training data: XOR problem
X = np.array([[0], [1], [0], [1]])
y = np.array([[0], [1], [1], [0]])

# Learning rate
lr = 0.5

print("Training simple network with backpropagation...\n")

for epoch in range(1000):
    total_loss = 0
    
    for i in range(len(X)):
        x = X[i]
        target = y[i]
        
        # === FORWARD PASS ===
        # Hidden layer
        z1 = np.dot(x, w1) + b1
        a1 = sigmoid(z1)
        
        # Output layer
        z2 = np.dot(a1, w2) + b2
        a2 = sigmoid(z2)
        
        # Compute loss (MSE)
        loss = (target - a2) ** 2
        total_loss += loss
        
        # === BACKWARD PASS ===
        # Output layer gradients
        d_loss_da2 = -2 * (target - a2)
        da2_dz2 = sigmoid_derivative(a2)
        d_loss_dz2 = d_loss_da2 * da2_dz2
        
        # Gradients for w2 and b2
        d_loss_dw2 = np.dot(a1.T, d_loss_dz2)
        d_loss_db2 = d_loss_dz2
        
        # Hidden layer gradients (chain rule!)
        d_loss_da1 = np.dot(d_loss_dz2, w2.T)
        da1_dz1 = sigmoid_derivative(a1)
        d_loss_dz1 = d_loss_da1 * da1_dz1
        
        # Gradients for w1 and b1
        d_loss_dw1 = np.dot(x.T, d_loss_dz1)
        d_loss_db1 = d_loss_dz1
        
        # === UPDATE WEIGHTS ===
        w2 -= lr * d_loss_dw2
        b2 -= lr * d_loss_db2
        w1 -= lr * d_loss_dw1
        b1 -= lr * d_loss_db1
    
    if epoch % 200 == 0:
        print(f"Epoch {epoch}: Loss = {total_loss/len(X):.6f}")

print("\n=== Final Predictions ===")
for i in range(len(X)):
    x = X[i]
    z1 = np.dot(x, w1) + b1
    a1 = sigmoid(z1)
    z2 = np.dot(a1, w2) + b2
    a2 = sigmoid(z2)
    print(f"Input: {x[0]} -> Output: {a2[0]:.4f} (Target: {y[i][0]})")
️ Ключевые моменты 1. Forward pass: вычисляем предсказание
2. Вычисляем ошибку (loss)
3. Backward pass: считаем градиенты через цепное правило
4. Обновляем веса в направлении анти-градиента

Векторизованный Backpropagation

Оптимизируем для скорости
+180 XP

Ручной backpropagation работает, но медленно — мы обрабатываем примеры по одному. В реальных сетях мы используем векторизацию: обрабатываем весь батч данных одновременно через матричные операции NumPy.

📝 Векторизованная версия

import numpy as np

class NeuralNetwork:
    def __init__(self, input_size, hidden_size, output_size):
        # Initialize weights with Xavier initialization
        self.w1 = np.random.randn(input_size, hidden_size) * np.sqrt(2.0 / input_size)
        self.b1 = np.zeros((1, hidden_size))
        self.w2 = np.random.randn(hidden_size, output_size) * np.sqrt(2.0 / hidden_size)
        self.b2 = np.zeros((1, output_size))
    
    def sigmoid(self, z):
        return 1 / (1 + np.exp(-np.clip(z, -500, 500)))
    
    def sigmoid_derivative(self, s):
        return s * (1 - s)
    
    def forward(self, X):
        """Forward pass for entire batch"""
        self.z1 = np.dot(X, self.w1) + self.b1
        self.a1 = self.sigmoid(self.z1)
        self.z2 = np.dot(self.a1, self.w2) + self.b2
        self.a2 = self.sigmoid(self.z2)
        return self.a2
    
    def backward(self, X, y, lr=0.5):
        """Backward pass and weight update"""
        m = X.shape[0]  # Batch size
        
        # Output layer gradients
        d_loss_da2 = -2 * (y - self.a2) / m
        da2_dz2 = self.sigmoid_derivative(self.a2)
        d_loss_dz2 = d_loss_da2 * da2_dz2
        
        d_loss_dw2 = np.dot(self.a1.T, d_loss_dz2)
        d_loss_db2 = np.sum(d_loss_dz2, axis=0, keepdims=True)
        
        # Hidden layer gradients
        d_loss_da1 = np.dot(d_loss_dz2, self.w2.T)
        da1_dz1 = self.sigmoid_derivative(self.a1)
        d_loss_dz1 = d_loss_da1 * da1_dz1
        
        d_loss_dw1 = np.dot(X.T, d_loss_dz1)
        d_loss_db1 = np.sum(d_loss_dz1, axis=0, keepdims=True)
        
        # Update weights
        self.w2 -= lr * d_loss_dw2
        self.b2 -= lr * d_loss_db2
        self.w1 -= lr * d_loss_dw1
        self.b1 -= lr * d_loss_db1
    
    def train(self, X, y, epochs=1000, lr=0.5):
        """Training loop"""
        for epoch in range(epochs):
            # Forward pass
            output = self.forward(X)
            
            # Compute loss
            loss = np.mean((y - output) ** 2)
            
            # Backward pass
            self.backward(X, y, lr)
            
            if epoch % 200 == 0:
                print(f"Epoch {epoch}: Loss = {loss:.6f}")
        
        return output

# Train on XOR problem
X = np.array([[0, 0], [0, 1], [1, 0], [1, 1]])
y = np.array([[0], [1], [1], [0]])

nn = NeuralNetwork(input_size=2, hidden_size=4, output_size=1)
final_output = nn.train(X, y, epochs=2000, lr=1.0)

print("\n=== Final Predictions ===")
for i in range(len(X)):
    print(f"Input: {X[i]} -> Output: {final_output[i][0]:.4f} (Target: {y[i][0]})")
📚 Xavier Initialization Мы используем Xavier initialization для весов: умножаем случайные числа на sqrt(2/input_size). Это помогает градиентам не затухать и не взрываться в глубоких сетях!

import numpy as np
import matplotlib.pyplot as plt

class DeepNeuralNetwork:
    """Full neural network with backpropagation"""
    
    def __init__(self, layer_sizes):
        """Initialize network with given layer sizes"""
        self.layer_sizes = layer_sizes
        self.weights = []
        self.biases = []
        
        # Xavier initialization
        for i in range(len(layer_sizes) - 1):
            w = np.random.randn(layer_sizes[i], layer_sizes[i+1]) * np.sqrt(2.0 / layer_sizes[i])
            b = np.zeros((1, layer_sizes[i+1]))
            self.weights.append(w)
            self.biases.append(b)
    
    def relu(self, z):
        return np.maximum(0, z)
    
    def relu_derivative(self, z):
        return (z > 0).astype(float)
    
    def sigmoid(self, z):
        return 1 / (1 + np.exp(-np.clip(z, -500, 500)))
    
    def sigmoid_derivative(self, s):
        return s * (1 - s)
    
    def forward(self, X):
        """Forward pass through all layers"""
        self.activations = [X]
        self.z_values = []
        
        current = X
        for i in range(len(self.weights) - 1):
            z = np.dot(current, self.weights[i]) + self.biases[i]
            self.z_values.append(z)
            current = self.relu(z)
            self.activations.append(current)
        
        # Output layer with sigmoid
        z = np.dot(current, self.weights[-1]) + self.biases[-1]
        self.z_values.append(z)
        output = self.sigmoid(z)
        self.activations.append(output)
        
        return output
    
    def backward(self, X, y, lr=0.01):
        """Backward pass and weight update"""
        m = X.shape[0]
        gradients_w = []
        gradients_b = []
        
        # Output layer gradient
        delta = (self.activations[-1] - y) * self.sigmoid_derivative(self.activations[-1])
        
        for i in range(len(self.weights) - 1, -1, -1):
            # Compute gradients
            dw = np.dot(self.activations[i].T, delta) / m
            db = np.sum(delta, axis=0, keepdims=True) / m
            
            gradients_w.insert(0, dw)
            gradients_b.insert(0, db)
            
            # Propagate error to previous layer
            if i > 0:
                delta = np.dot(delta, self.weights[i].T) * self.relu_derivative(self.z_values[i-1])
        
        # Update weights
        for i in range(len(self.weights)):
            self.weights[i] -= lr * gradients_w[i]
            self.biases[i] -= lr * gradients_b[i]
    
    def train(self, X, y, epochs=1000, lr=0.01, verbose=True):
        """Training loop with loss tracking"""
        losses = []
        
        for epoch in range(epochs):
            # Forward pass
            output = self.forward(X)
            
            # Compute loss
            loss = -np.mean(y * np.log(output + 1e-8) + (1 - y) * np.log(1 - output + 1e-8))
            losses.append(loss)
            
            # Backward pass
            self.backward(X, y, lr)
            
            if verbose and epoch % 100 == 0:
                accuracy = np.mean((output > 0.5) == y)
                print(f"Epoch {epoch}: Loss = {loss:.4f}, Accuracy = {accuracy:.2%}")
        
        return losses

# Train on spiral dataset
np.random.seed(42)
N = 200

# Create spiral data
t = np.linspace(0, 2*np.pi, N)
r = np.linspace(0.5, 2, N)
X0 = np.column_stack([r*np.cos(t) + np.random.randn(N)*0.1, 
                      r*np.sin(t) + np.random.randn(N)*0.1])
