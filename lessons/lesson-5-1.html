Почему RNN больше не нужны?

Проблема длинных зависимостей
+100 XP

До 2017 года для работы с текстом использовали RNN (рекуррентные нейросети) и LSTM. Они читали текст слово за словом, запоминая контекст. Но у них была проблема: чем длиннее текст, тем хуже они помнили начало. Как человек, который читает книгу и забывает первую главу к концу!

📖
Игровая аналогия RNN — это как ученик, который читает текст слева направо и пытается запомнить всё по порядку. Если текст длинный, он забывает начало. Transformer — это как гений, который может одновременно смотреть на все слова и понимать связи между любыми из них, независимо от расстояния!

👁️ Что такое Attention?

Механизм внимания
+200 XP

Attention (внимание) — это механизм, который позволяет модели "смотреть" на разные части входного текста с разной интенсивностью. Когда ты читаешь предложение "Кот сидит на коврике, потому что он тёплый", слово "он" относится к "коврику", а не к "коту". Attention помогает модели понять это!

📝 Интуиция Attention на примере

# Imagine you're translating: "The animal didn't cross the street because it was too tired"
# What does "it" refer to? "animal" or "street"?

# Attention mechanism looks at all words and assigns weights
sentence = ["The", "animal", "didn't", "cross", "the", "street", "because", "it", "was", "too", "tired"]

# When processing "it", attention focuses on relevant words
attention_weights = {
    "The": 0.02,
    "animal": 0.35,      # High attention - "it" refers to animal
    "didn't": 0.05,
    "cross": 0.08,
    "the": 0.02,
    "street": 0.15,
    "because": 0.10,
    "it": 0.00,          # Can't attend to itself
    "was": 0.05,
    "too": 0.08,
    "tired": 0.10        # "tired" confirms it's about animal
}

print("Attention weights when processing 'it':")
for word, weight in sorted(attention_weights.items(), key=lambda x: -x[1]):
    bar = "█" * int(weight * 50)
    print(f"  {word:10s}: {weight:.2f} {bar}")

print("\nThe model correctly understands 'it' = 'animal'!")
💡 Ключевая идея Attention позволяет модели динамически фокусироваться на разных частях входа при обработке каждого токена. Это как иметь суперспособность читать все слова одновременно и понимать связи между ними!

🔑 Query, Key, Value

Математика Self-Attention
+250 XP

Self-Attention работает через три вектора для каждого токена: Query (запрос) — что я ищу? Key (ключ) — что я могу предложить? Value (значение) — какая у меня информация? Attention score = similarity(Query, Key), и мы взвешиваем Values по этим scores.

️
Игровая аналогия Представь библиотеку: Query — это твой запрос ("хочу книгу о драконах"), Key — это метки на книгах ("фэнтези", "драконы", "магия"), Value — это содержимое книг. Attention находит книги с ключами, похожими на твой запрос, и возвращает их содержимое, взвешенное по релевантности!

📝 Реализация Self-Attention

import torch
import torch.nn as nn
import torch.nn.functional as F

class SelfAttention(nn.Module):
    """Self-Attention mechanism implementation"""
    
    def __init__(self, embed_size, num_heads=1):
        super().__init__()
        self.embed_size = embed_size
        self.num_heads = num_heads
        
        # Linear transformations for Q, K, V
        self.query = nn.Linear(embed_size, embed_size)
        self.key = nn.Linear(embed_size, embed_size)
        self.value = nn.Linear(embed_size, embed_size)
    
    def forward(self, x):
        # x shape: (batch_size, seq_length, embed_size)
        batch_size, seq_length, embed_size = x.shape
        
        # Compute Q, K, V
        Q = self.query(x)  # (batch, seq, embed)
        K = self.key(x)    # (batch, seq, embed)
        V = self.value(x)  # (batch, seq, embed)
        
        # Calculate attention scores
        # scores shape: (batch, seq, seq)
        scores = torch.matmul(Q, K.transpose(-2, -1)) / (embed_size ** 0.5)
        
        # Apply softmax to get attention weights
        attention_weights = F.softmax(scores, dim=-1)
        
        # Weighted sum of values
        # output shape: (batch, seq, embed)
        output = torch.matmul(attention_weights, V)
        
        return output, attention_weights

# Test the attention mechanism
embed_size = 64
seq_length = 10
batch_size = 2

attention = SelfAttention(embed_size)
x = torch.randn(batch_size, seq_length, embed_size)

output, weights = attention(x)
print(f"Input shape: {x.shape}")
print(f"Output shape: {output.shape}")
print(f"Attention weights shape: {weights.shape}")
print(f"\nAttention weights for first token (first 5 values):")
print(weights[0, 0, :5])
📚 Почему делим на sqrt(d)? Масштабирование на d_k (размерность ключа) предотвращает слишком большие значения dot product, которые могли бы привести к очень маленьким градиентам после softmax (проблема vanishing gradients).

🎯 Multi-Head Attention

Несколько голов внимания
+300 XP

Вместо одного Attention, Transformer использует Multi-Head Attention — несколько "голов" внимания параллельно. Каждая голова учится фокусироваться на разных аспектах: одна на синтаксисе, другая на семантике, третья на позициях слов. Потом результаты объединяются!

📝 Multi-Head Attention в PyTorch

import torch
import torch.nn as nn

class MultiHeadAttention(nn.Module):
    """Multi-Head Attention mechanism"""
    
    def __init__(self, embed_size, num_heads):
        super().__init__()
        self.embed_size = embed_size
        self.num_heads = num_heads
        self.head_dim = embed_size // num_heads
        
        assert self.head_dim * num_heads == embed_size, "embed_size must be divisible by num_heads"
        
        # Linear layers for Q, K, V
        self.qkv = nn.Linear(embed_size, embed_size * 3)
        self.fc_out = nn.Linear(embed_size, embed_size)
    
    def forward(self, x):
        batch_size, seq_length, embed_size = x.shape
        
        # Compute Q, K, V and split into heads
        qkv = self.qkv(x).reshape(batch_size, seq_length, 3, self.num_heads, self.head_dim)
        qkv = qkv.permute(2, 0, 3, 1, 4)  # (3, batch, heads, seq, head_dim)
        Q, K, V = qkv[0], qkv[1], qkv[2]
        
        # Scaled dot-product attention
        energy = torch.matmul(Q, K.transpose(-2, -1)) / (self.head_dim ** 0.5)
        attention = torch.softmax(energy, dim=-1)
        
        # Apply attention to values
        out = torch.matmul(attention, V)
        
        # Concatenate heads
        out = out.reshape(batch_size, seq_length, embed_size)
        out = self.fc_out(out)
        
        return out

# Test Multi-Head Attention
embed_size = 256
num_heads = 8
mha = MultiHeadAttention(embed_size, num_heads)

x = torch.randn(2, 10, embed_size)  # batch=2, seq=10
output = mha(x)

print(f"Input shape: {x.shape}")
print(f"Output shape: {output.shape}")
print(f"Number of attention heads: {num_heads}")
print(f"Dimension per head: {embed_size // num_heads}")
💡 Почему несколько голов? Разные головы могут учиться разным типам зависимостей: одна голова следит за грамматикой, другая — за смыслом, третья — за позицией слов. Это как иметь команду экспертов, каждый из которых смотрит на текст под своим углом!

🏗️ Архитектура Transformer

Encoder-Decoder структура
+350 XP

Полный Transformer состоит из Encoder (кодирует вход) и Decoder (генерирует выход). Каждый содержит слои Multi-Head Attention, Feed-Forward сети, Layer Normalization и residual connections. GPT использует только Decoder, BERT — только Encoder, а T5 — оба!

Transformer Encoder Layer

import torch
import torch.nn as nn

class TransformerEncoderLayer(nn.Module):
    """Single Transformer Encoder Layer"""
    
    def __init__(self, embed_size, num_heads, ff_dim, dropout=0.1):
        super().__init__()
        self.attention = nn.MultiheadAttention(embed_size, num_heads, dropout=dropout)
        self.norm1 = nn.LayerNorm(embed_size)
        self.norm2 = nn.LayerNorm(embed_size)
        
        # Feed-Forward Network
        self.ff = nn.Sequential(
            nn.Linear(embed_size, ff_dim),
            nn.ReLU(),
            nn.Dropout(dropout),
            nn.Linear(ff_dim, embed_size),
            nn.Dropout(dropout)
        )
        
        self.dropout = nn.Dropout(dropout)
    
    def forward(self, x):
        # Self-Attention with residual connection
        attn_output, _ = self.attention(x, x, x)
        x = self.norm1(x + self.dropout(attn_output))
        
        # Feed-Forward with residual connection
        ff_output = self.ff(x)
        x = self.norm2(x + self.dropout(ff_output))
        
        return x

class TransformerEncoder(nn.Module):
    """Full Transformer Encoder with multiple layers"""
    
    def __init__(self, vocab_size, embed_size, num_layers, num_heads, ff_dim, max_seq_length, dropout=0.1):
        super().__init__()
        self.embed_size = embed_size
        
        # Token and Position Embeddings
        self.token_embedding = nn.Embedding(vocab_size, embed_size)
        self.pos_embedding = nn.Embedding(max_seq_length, embed_size)
        
        # Stack of encoder layers
        self.layers = nn.ModuleList([
            TransformerEncoderLayer(embed_size, num_heads, ff_dim, dropout)
            for _ in range(num_layers)
        ])
        
        self.dropout = nn.Dropout(dropout)
    
    def forward(self, x):
        # x shape: (batch_size, seq_length)
        batch_size, seq_length = x.shape
        positions = torch.arange(seq_length, device=x.device).unsqueeze(0)
        
        # Add token and position embeddings
        x = self.token_embedding(x) + self.pos_embedding(positions)
        x = self.dropout(x)
        
        # Pass through encoder layers
        for layer in self.layers:
            x = layer(x)
        
        return x

# Create a small Transformer
vocab_size = 10000
embed_size = 256
num_layers = 6
num_heads = 8

max_seq_length = 512

transformer = TransformerEncoder(vocab_size, embed_size, num_layers, num_heads, ff_dim, max_seq_length)

# Test with dummy input

output = transformer(x)

print(f"Total parameters: {sum(p.numel() for p in transformer.parameters()):,}")

📝 Мини-GPT: Decoder-only Transformer

import torch
import torch.nn as nn
import torch.nn.functional as F

class CausalSelfAttention(nn.Module):
    """Self-Attention with causal mask (can't see future tokens)"""
    
    def __init__(self, embed_size, num_heads):
        super().__init__()
        self.num_heads = num_heads
        self.head_dim = embed_size // num_heads
        
        self.qkv = nn.Linear(embed_size, embed_size * 3)
        self.proj = nn.Linear(embed_size, embed_size)
        
        # Causal mask (lower triangular)
        self.register_buffer('mask', torch.tril(torch.ones(1024, 1024)).view(1, 1, 1024, 1024))
    
    def forward(self, x):
        B, T, C = x.shape
        
        # Compute Q, K, V
        qkv = self.qkv(x).reshape(B, T, 3, self.num_heads, self.head_dim)
        qkv = qkv.permute(2, 0, 3, 1, 4)
        q, k, v = qkv[0], qkv[1], qkv[2]
        
        # Attention scores
        att = (q @ k.transpose(-2, -1)) * (1.0 / (k.size(-1) ** 0.5))
        
        # Apply causal mask
        att = att.masked_fill(self.mask[:, :, :T, :T] == 0, float('-inf'))
        att = F.softmax(att, dim=-1)
        
        # Apply attention to values
        y = att @ v
        y = y.transpose(1, 2).contiguous().view(B, T, C)
        
        return self.proj(y)

class MiniGPTBlock(nn.Module):
    """Single GPT block (Decoder layer)"""
    
    def __init__(self, embed_size, num_heads):
        super().__init__()
        self.ln1 = nn.LayerNorm(embed_size)
        self.attn = CausalSelfAttention(embed_size, num_heads)
        self.ln2 = nn.LayerNorm(embed_size)
        self.ff = nn.Sequential(
            nn.Linear(embed_size, 4 * embed_size),
            nn.GELU(),
            nn.Linear(4 * embed_size, embed_size)
        )
    
    def forward(self, x):
        x = x + self.attn(self.ln1(x))
        x = x + self.ff(self.ln2(x))
        return x

class MiniGPT(nn.Module):
    """Mini GPT language model"""
    
    def __init__(self, vocab_size, embed_size, num_layers, num_heads, max_seq_length):
        super().__init__()
        self.token_embed = nn.Embedding(vocab_size, embed_size)
        self.pos_embed = nn.Embedding(max_seq_length, embed_size)
        self.blocks = nn.Sequential(*[MiniGPTBlock(embed_size, num_heads) for _ in range(num_layers)])
        self.ln_f = nn.LayerNorm(embed_size)
        self.head = nn.Linear(embed_size, vocab_size, bias=False)
    
    def forward(self, idx, targets=None):
        B, T = idx.shape
        pos = torch.arange(T, device=idx.device).unsqueeze(0)
        
        x = self.token_embed(idx) + self.pos_embed(pos)
        x = self.blocks(x)
        x = self.ln_f(x)
        logits = self.head(x)
        
        loss = None
        if targets is not None:
            loss = F.cross_entropy(logits.view(-1, logits.size(-1)), targets.view(-1))
        
        return logits, loss

# Create mini GPT

embed_size = 128
num_layers = 4
num_heads = 4
max_seq_length = 64

model = MiniGPT(vocab_size, embed_size, num_layers, num_heads, max_seq_length)

print("=== Mini GPT Architecture ===")
print(f"Vocabulary size: {vocab_size}")

# Test forward pass

# Generate text
def generate(model, idx, max_new_tokens, temperature=1.0):
    model.eval()
    for _ in range(max_new_tokens):
        idx_cond = idx[:, -max_seq_length:]
        logits, _ = model(idx_cond)
        logits = logits[:, -1, :] / temperature
        probs = F.softmax(logits, dim=-1)
        idx_next = torch.multinomial(probs, num_samples=1)
        idx = torch.cat([idx, idx_next], dim=1)
    return idx

print("\n=== Text Generation Demo ===")
start_tokens = torch.randint(0, vocab_size, (1, 10))
generated = generate(model, start_tokens, max_new_tokens=20)
print(f"Generated token IDs: {generated[0].tolist()}")
print("Note: With random weights, output is random. After training, it would generate meaningful text!")
