🔬 Подходы к анализу тональности

От словарей до нейросетей
+150 XP

Существует три основных подхода: 1) Словарный — подсчёт "хороших" и "плохих" слов; 2) Классический ML — обучение классификатора на признаках; 3) Deep Learning — использование предобученных трансформеров (BERT, RoBERTa). Сегодня мы используем третий — самый мощный!

📝 Сравнение подходов

# Approach 1: Dictionary-based (simple but limited)
positive_words = {'good', 'great', 'excellent', 'amazing', 'love'}
negative_words = {'bad', 'terrible', 'awful', 'hate', 'horrible'}

def dictionary_sentiment(text):
    """Simple sentiment analysis using word lists"""
    words = set(text.lower().split())
    pos_count = len(words & positive_words)
    neg_count = len(words & negative_words)
    
    if pos_count > neg_count:
        return "positive", pos_count - neg_count
    elif neg_count > pos_count:
        return "negative", neg_count - pos_count
    return "neutral", 0

# Test it
print(dictionary_sentiment("This movie is good and great"))  # positive
print(dictionary_sentiment("This is terrible and awful"))    # negative

# Problem: doesn't understand context!
print(dictionary_sentiment("Not good at all"))  # Still says positive! 😅
⚠️ Проблема словарного подхода Словарь не понимает контекст, иронию, отрицания ("not good" = плохо, но словарь видит "good" = хорошо). Поэтому современные решения используют нейросети!

🤗 HuggingFace Pipeline

Анализ тональности в 3 строки кода
+200 XP

Библиотека HuggingFace Transformers предоставляет готовые pipeline'ы — функции, которые делают всю сложную работу за нас. Для анализа тональности достаточно 3 строк кода!

📝 Sentiment Analysis с HuggingFace

from transformers import pipeline

# Load pre-trained sentiment analysis pipeline
# This downloads a model trained on thousands of reviews
sentiment_analyzer = pipeline("sentiment-analysis")

# Test with different texts
texts = [
    "I absolutely love this product! Best purchase ever.",
    "This is the worst experience I've ever had.",
    "The movie was okay, nothing special.",
    "Terrible service, would not recommend to anyone!",
    "Amazing quality and fast shipping. 5 stars!"
]

print("=== Sentiment Analysis Results ===\n")
for text in texts:
    result = sentiment_analyzer(text)[0]
    label = result['label']
    score = result['score']
    emoji = "😊" if label == "POSITIVE" else "" if label == "NEGATIVE" else ""
    
    print(f"{emoji} Text: {text}")
    print(f"   Sentiment: {label} (confidence: {score:.2%})")
    print()
💡 Как это работает? Pipeline автоматически загружает модель (обычно DistilBERT), токенизирует текст, пропускает через нейросеть и возвращает результат. Вся магия из урока 4.2 происходит "под капотом"!

🚨 Детекция токсичности

Распознавание оскорблений и угроз
+250 XP

Обычный sentiment analysis определяет "позитив/негатив", но для модерации нам нужна детекция токсичности — распознавание оскорблений, угроз, ненависти. Для этого есть специальные модели, например Detoxify, обученные на миллионах помеченных комментариев.

📝 Детекция токсичности с Detoxify

# Install first: pip install detoxify
from detoxify import Detoxify

# Load the multilingual toxicity model
model = Detoxify('multilingual')  # Supports English, Russian, etc.

# Test comments (some toxic, some not)
comments = [
    "Great video, thanks for sharing!",
    "You're an idiot and your content is trash",
    "I disagree with your opinion, but respect it",
    "Go kill yourself, nobody likes you",
    "This tutorial helped me a lot, thank you!",
    "Stupid waste of time, worst video ever"
]

print("=== Toxicity Detection ===\n")
for comment in comments:
    results = model.predict(comment)
    
    # Check if any toxicity category is high
    is_toxic = any(score > 0.5 for score in results.values())
    
    status = "🚨 TOXIC" if is_toxic else "✅ SAFE"
    print(f"{status} | {comment}")
    print(f"   Scores:")
    for category, score in results.items():
        bar = "█" * int(score * 20)
        print(f"   {category:15s}: {score:.3f} {bar}")
    print()
📚 Категории токсичности Detoxify определяет 6 типов токсичности: toxicity (общая), severe_toxicity (тяжёлая), obscene (нецензурная), identity_attack (against identity), insult (оскорбление), threat (угроза). Для бана обычно достаточно toxicity > 0.7!

Полный код AI-модератора

import os
import logging
from dotenv import load_dotenv
from telegram import Update
from telegram.ext import Application, MessageHandler, filters, ContextTypes
from detoxify import Detoxify

# Load environment variables
load_dotenv()
BOT_TOKEN = os.getenv('BOT_TOKEN')
ALLOWED_CHAT_ID = int(os.getenv('ALLOWED_CHAT_ID', 0))

# Setup logging
logging.basicConfig(
    format='%(asctime)s - %(name)s - %(levelname)s - %(message)s',
    level=logging.INFO
)
logger = logging.getLogger(__name__)

# Initialize toxicity model (loads once at startup)
logger.info("Loading toxicity model...")
toxicity_model = Detoxify('multilingual')
logger.info("Model loaded successfully!")

# User warning counter
user_warnings = {}
MAX_WARNINGS = 3

async def check_toxicity(text: str) -> dict:
    """Analyze text for toxicity and return scores"""
    try:
        results = toxicity_model.predict(text)
        return results
    except Exception as e:
        logger.error(f"Toxicity check failed: {e}")
        return {}

async def is_toxic(scores: dict, threshold: float = 0.7) -> bool:
    """Check if any toxicity score exceeds threshold"""
    return any(score > threshold for score in scores.values())

async def message_handler(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """Handle incoming messages and check for toxicity"""
    # Only process messages in allowed chat
    if update.effective_chat.id != ALLOWED_CHAT_ID:
        return
    
    # Skip messages from bots
    if update.message.from_user.is_bot:
        return
    
    message_text = update.message.text
    user_id = update.message.from_user.id
    username = update.message.from_user.username or str(user_id)
    
    # Skip commands
    if message_text.startswith('/'):
        return
    
    logger.info(f"Checking message from @{username}: {message_text[:50]}...")
    
    # Analyze toxicity
    scores = await check_toxicity(message_text)
    
    if not scores:
        return
    
    # Check toxicity level
    max_score = max(scores.values())
    
    if max_score > 0.9:
        # Severe toxicity - immediate ban
        await update.message.reply_text(
            f" @{username} был забанен за серьёзное нарушение правил.\n"
            f"Уровень токсичности: {max_score:.1%}"
        )
        await context.bot.ban_chat_member(
            chat_id=update.effective_chat.id,
            user_id=user_id
        )
        logger.warning(f"Banned @{username} (score: {max_score:.2f})")
        
    elif max_score > 0.7:
        # High toxicity - warning
        user_warnings[user_id] = user_warnings.get(user_id, 0) + 1
        warnings = user_warnings[user_id]
        
        await update.message.reply_text(
            f"️ @{username}, предупреждение {warnings}/{MAX_WARNINGS}!\n"
            f"Твоё сообщение нарушает правила сообщества.\n"
            f"Уровень токсичности: {max_score:.1%}\n"
            f"При {MAX_WARNINGS} предупреждениях ты будешь забанен."
        )
        logger.warning(f"Warning #{warnings} for @{username}")
        
        # Ban if max warnings reached
        if warnings >= MAX_WARNINGS:
            await update.message.reply_text(
                f"❌ @{username} забанен после {MAX_WARNINGS} предупреждений."
            )
            await context.bot.ban_chat_member(
                chat_id=update.effective_chat.id,
                user_id=user_id
            )
            user_warnings[user_id] = 0
            logger.warning(f"Banned @{username} after {MAX_WARNINGS} warnings")
    
    elif max_score > 0.5:
        # Mild toxicity - just log
        logger.info(f"Mild toxicity from @{username}: {max_score:.2f}")

async def start_command(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """Handle /start command"""
    await update.message.reply_text(
        "🤖 AI-модератор активирован!\n\n"
        "Я буду проверять все сообщения на токсичность.\n"
        "Уровень токсичности > 70% = предупреждение\n"
        "Уровень токсичности > 90% = бан\n"
        "3 предупреждения = бан\n\n"
        "Давайте поддерживать дружелюбную атмосферу! 😊"
    )

async def stats_command(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """Handle /stats command"""
    total_warnings = sum(user_warnings.values())
    await update.message.reply_text(
        f"📊 Статистика модерации:\n"
        f"Активных предупреждений: {total_warnings}\n"
        f"Модель: Detoxify Multilingual\n"
        f"Статус: ✅ Активен"
    )

def main():
    """Start the bot"""
    if not BOT_TOKEN:
        logger.error("BOT_TOKEN not found in .env file!")
        return
    
    logger.info("Starting AI Moderator Bot...")
    
    # Create application
    application = Application.builder().token(BOT_TOKEN).build()
    
    # Register handlers
    application.add_handler(CommandHandler("start", start_command))
    application.add_handler(CommandHandler("stats", stats_command))
    application.add_handler(
        MessageHandler(filters.TEXT & ~filters.COMMAND, message_handler)
    )
    
    # Start polling
    logger.info("Bot is running...")
    application.run_polling(allowed_updates=Update.ALL_TYPES)

if __name__ == '__main__':
    main()
